## 二、几何不变量（可机器校验，本版全部通过）

用两条等式取代"目测居中"：

- **水平**：`x0 + x1 = 15` → 保证 cx = 7.5
- **垂直**：`y0 + y1 = 13` → 保证 cy = 6.5（可用带只有 rows 0–13）

推论：bbox 宽高必须是**偶数**。奇数尺寸的图形在 16 格里数学上无法居中，这是前几版反复偏 1px 的根因。另外三条：r14/r15 全透明、alpha 仅 0/255、像素集四连通。

Alpha is only 0 or 255. #(68) and @(24) are unused: on the tooltip
background (#100010, alpha 240) they are indistinguishable from empty.
"""

# ---------------- validation ----------------

def validate(name, grid):
    errs = []
    if len(grid) != 16:
        errs.append(f"{len(grid)} rows, expected 16")
    for y, row in enumerate(grid):
        if len(row) != 16:
            errs.append(f"r{y}: {len(row)} chars, expected 16")
        for x, ch in enumerate(row):
            if ch != '.' and ch not in PAL:
                errs.append(f"r{y}c{x}: unknown char {ch!r}")
    pts = {(y, x) for y, row in enumerate(grid)
           for x, ch in enumerate(row) if ch != '.'}
    if not pts:
        return errs + ["empty grid"]
    ys, xs = [p[0] for p in pts], [p[1] for p in pts]
    if max(ys) > 13:
        errs.append("bleeds into r14/r15 (collides with the next tooltip line)")
    if min(xs) + max(xs) != 15:
        errs.append(f"x0+x1={min(xs)+max(xs)} != 15 (cx={(min(xs)+max(xs))/2})")
    if min(ys) + max(ys) != 13:
        errs.append(f"y0+y1={min(ys)+max(ys)} != 13 (cy={(min(ys)+max(ys))/2})")
    seen, q = {next(iter(pts))}, deque([next(iter(pts))])
    while q:
        y, x = q.popleft()
        for n in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
            if n in pts and n not in seen:
                seen.add(n)
                q.append(n)
    if len(seen) != len(pts):
        orphans = sorted(pts - seen)
        errs.append(f"not 4-connected: {len(orphans)} detached px, first {orphans[0]}")
    return errs

def audit(name, grid):
    pts = [(y, x, ch) for y, row in enumerate(grid)
           for x, ch in enumerate(row) if ch != '.']
    ys, xs = [p[0] for p in pts], [p[1] for p in pts]
    x0, x1, y0, y1 = min(xs), max(xs), min(ys), max(ys)
    w = sum(LUM[p[2]] for p in pts) / 255
    print(f"{name:14s} cols {x0:2d}-{x1:2d}  rows {y0:2d}-{y1:2d}  "
          f"cx={(x0+x1)/2:4.1f} cy={(y0+y1)/2:4.1f}  "
          f"{len(pts):2d}px  weight={w:5.1f}")

# ---------------- output ----------------

def render(grid):
    return [PAL.get(ch, (0, 0, 0, 0)) for row in grid for ch in row]

def write_png(path, px, w=16, h=16):
    assert len(px) == w * h, f"{len(px)} px for {w}x{h}"
    def chunk(tag, data):
        return (struct.pack('>I', len(data)) + tag + data
                + struct.pack('>I', zlib.crc32(tag + data) & 0xffffffff))
    body, stride = bytes(v for p in px for v in p), w * 4
    raw = b''.join(b'\x00' + body[i:i + stride] for i in range(0, len(body), stride))
    with open(path, 'wb') as f:
        f.write(b'\x89PNG\r\n\x1a\n'
                + chunk(b'IHDR', struct.pack('>IIBBBBB', w, h, 8, 6, 0, 0, 0))
                + chunk(b'IDAT', zlib.compress(raw, 9))
                + chunk(b'IEND', b''))

def write_sheet(path, names, scale=8):
    """Vertical stack at the real 14px line pitch on the real tooltip colour."""
    cols, rows = 20, 14 * len(names) + 2
    canvas = [TOOLTIP_BG] * (cols * rows)
    for i, n in enumerate(names):
        px = render(ICONS[n])
        for y in range(16):
            for x in range(16):
                p = px[y * 16 + x]
                ty = i * 14 + y
                if p[3] and ty < rows:
                    canvas[ty * cols + x] = p
    W, H = cols * scale, rows * scale
    big = [canvas[(y // scale) * cols + (x // scale)]
           for y in range(H) for x in range(W)]
    write_png(path, big, W, H)

if __name__ == "__main__":
    names = sys.argv[1:] or list(ICONS)
    failed = False
    for n in names:
        if n not in ICONS:
            sys.exit(f"unknown icon: {n} (have: {', '.join(ICONS)})")
        errs = validate(n, ICONS[n])
        for e in errs:
            print(f"FAIL {n}: {e}")
        failed |= bool(errs)
    if failed:
        sys.exit(1)
    for n in names:
        audit(n, ICONS[n])
        write_png(os.path.join(OUT, n + ".png"), render(ICONS[n]))
    write_sheet(os.path.join(OUT, "_sheet.png"), names)
    print(f"\nwrote {len(names)} icons + _sheet.png -> {OUT}")

## 六、已核验 / 未核验

**已核验**（对 6 个网格逐格手算，且 `validate()` 会在构建时复查）：每行恰 16 字符、字符全在 `PAL` 内、r14/r15 全透明、`x0+x1=15` 且 `y0+y1=13`、像素集四连通无孤立点、`#`(68) 与 `@`(24) 已从全套移除。

**未核验**：观感。我没有渲染能力，这必须你跑一次 `_sheet.png` 才能定。

**唯一已知缺口**：靴子光学重量 19.9，仅为基准（34）的 58%，竖列里会明显偏弱。根因是原版皮革配色本身暗 —— `t`=147 / `T`=95 / `b`=55 感知亮度，铁灰是 255/190/107。47px 面积配这套色，数学上到不了 34。三条出路：

| 方案 | 重量 | 代价 |
|---|---|---|
| 保持现状 | 19.9 | 移速那行偏弱（该属性也最不重要） |
| 加一档亮皮革 `u`=(215,170,120)，`t`→本体、`u`→高光 | ≈25 | 略偏离原版皮革靴配色，仍在同色系 |
| 换铁靴（`W`/`M`/`d`） | ≈34 | 与剑/胸甲同为铁灰，6 行色相区分度下降 |

我的建议是先按现状出图，看 `_sheet.png` 里靴子是否真的"沉下去"再决定 —— 亮度加权是估算，实际感知还受色相和周围元素影响。
