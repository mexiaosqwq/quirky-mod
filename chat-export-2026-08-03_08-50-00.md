```python
def write_sheet(path, icons, scale=8, gap=2, bg=(16, 0, 16, 255)):
    """Vertical column on the real tooltip background, at display-ish scale."""
    cols, rows = 16, 14 * len(icons) + 2
    canvas = [bg] * (cols * rows)
    for i, grid in enumerate(icons):
        px = render(grid)
        for y in range(16):
            for x in range(16):
                p = px[y * 16 + x]
                if p[3]:
                    ty = i * 14 + y
                    if 0 <= ty < rows:
                        canvas[ty * cols + x] = p
    big, W, H = [], cols * scale, rows * scale
    for y in range(H):
        row = [canvas[(y // scale) * cols + (x // scale)] for x in range(W)]
        big.extend(row)
    _write_png_size(path, big, W, H)   # write_png 里把 16 换成 W/H 即可
```
